function moveField(btn,dir){
  const row=btn.closest('.q-node');
  const list=document.getElementById('signFieldList');
  const rows=Array.from(list.querySelectorAll('.q-node'));
  const i=rows.indexOf(row);

/* 演示用二维码：按项目名生成稳定的伪二维码图案（真实环境由服务端生成） */
function renderQR(seedStr){
  let seed=0; for(let i=0;i<seedStr.length;i++){ seed=(seed*31+seedStr.charCodeAt(i))>>>0; }
  const rnd=()=>{ seed=(seed*1664525+1013904223)>>>0; return seed/4294967296; };
  const N=25, cell=100/N;
  let rects='';
  const finder=(x,y)=>{
    rects+='<rect x="'+x*cell+'" y="'+y*cell+'" width="'+cell*7+'" height="'+cell*7+'" fill="#111"/>';
    rects+='<rect x="'+(x+1)*cell+'" y="'+(y+1)*cell+'" width="'+cell*5+'" height="'+cell*5+'" fill="#fff"/>';
    rects+='<rect x="'+(x+2)*cell+'" y="'+(y+2)*cell+'" width="'+cell*3+'" height="'+cell*3+'" fill="#111"/>';
  };
  for(let y=0;y<N;y++){
    for(let x=0;x<N;x++){
      const inFinder=(x<8&&y<8)||(x>N-9&&y<8)||(x<8&&y>N-9);
      if(!inFinder&&rnd()>0.52){ rects+='<rect x="'+x*cell+'" y="'+y*cell+'" width="'+cell+'" height="'+cell+'" fill="#111"/>'; }
    }
  }
  finder(0,0); finder(N-7,0); finder(0,N-7);
  document.getElementById('promo-qr').innerHTML='<svg viewBox="0 0 100 100" xmlns="http://www.w3.org/2000/svg"><rect width="100" height="100" fill="#fff"/>'+rects+'</svg>';
}
function downloadQR(){
  const svg=document.querySelector('#promo-qr svg');
  if(!svg) return;
  const blob=new Blob([svg.outerHTML],{type:'image/svg+xml'});
  const a=document.createElement('a');
  a.href=URL.createObjectURL(blob);
  a.download='推广二维码_'+promoCur+'.svg';
  a.click();
  URL.revokeObjectURL(a.href);
  alert('二维码已下载（演示，SVG格式）：'+promoCur);
}
