let rjCount=0, rjNames=[];
function batchReject(){
  const sel=getChecked('zu-ju');
  if(!sel.length){ alert('请先勾选审核不通过的志愿者'); return; }
  rjCount=sel.length;
  rjNames=[];
  sel.forEach(c=>{
    const tr=c.closest('tr');
    const nm=tr.querySelectorAll('td')[2];
    rjNames.push(nm.textContent.split('\n')[0].trim());
  });
  document.getElementById('rj-title').textContent='批量审核不通过（已选 '+rjCount+' 人）';
  document.querySelectorAll('#rejectMask .rj-opt').forEach(o=>o.classList.remove('on'));
  document.getElementById('rj-note').value='';
  document.getElementById('rejectMask').classList.add('show');
}

function saveReject(){
  const reasons=Array.from(document.querySelectorAll('#rejectMask .rj-opt.on')).map(o=>o.textContent);
  if(!reasons.length){ alert('请至少选择一个不通过原因'); return; }

  closeRejectMask();

    +(note?('\n备注：'+note):'')
    +'\n志愿者收到未入选通知；可在列表行「编辑原因」继续调整');
  getChecked('zu-ju').forEach(c=>c.checked=false);
  updateSelCount('zu-ju','selCount-1');
}

function searchCkProj(){
  const kw=document.getElementById('ckProjSearch').value.trim();
  let tip=document.getElementById('ppHistoryTip');
  if(!kw){ if(tip) tip.style.display='none'; alert('请输入项目名称关键词'); return; }
  const hits=ckProjAll.filter(p=>p.name.indexOf(kw)>-1);
  if(!tip){
    tip=document.createElement('div');
    tip.className='pp-history-tip';
    tip.id='ppHistoryTip';
    const picker=document.querySelector('#ckProjTabs').closest('.proj-picker');
    picker.appendChild(tip);
  }
  if(!hits.length){
    tip.innerHTML='未找到匹配「'+kw.replace(/</g,'&lt;')+'」的项目，请更换关键词';
    tip.style.display='block';
    return;
  }
  tip.innerHTML='<b>搜索结果（点击切换项目查看打卡数据）：</b>'+hits.map(p=>
    '<div class="h-item"><span onclick="useCkProj(\''+p.name+'\',\''+p.sub+'\')">'+p.name+' · '+p.sub+'</span></div>').join('');
  tip.style.display='block';
}
function useCkProj(name,sub){
  const tabs=document.getElementById('ckProjTabs');
  tabs.querySelectorAll('span').forEach(s=>s.classList.remove('on'));
  let hit=null;
  tabs.querySelectorAll('span').forEach(s=>{ if(s.textContent.indexOf(name)>-1){ hit=s; } });
  if(hit){ hit.classList.add('on'); }
  else{
    const sp=document.createElement('span');
    sp.innerHTML=name+'<span class="pp-sub">'+sub+'</span>';
    sp.onclick=function(){ pickCkProj(sp); };
    tabs.appendChild(sp);
  }
  document.getElementById('ppHistoryTip').style.display='none';
}
