/* ===== 报名草稿（防止跳转简道云签署后丢失） ===== */
const draft={proj:null,mt:null,skin:null,sens:null};
function saveDraft(){
  draft.proj=document.getElementById('f-proj').value;
  draft.mt=document.getElementById('f-mt').value;
  const s=document.querySelector('#skinGrid span.on'); draft.skin=s?s.textContent:null;
  const n=document.querySelector('#sensGrid span.on'); draft.sens=n?n.textContent:null;
}
function restoreDraft(){
  const has=draft.proj||draft.mt||draft.skin||draft.sens;
  const tip=document.getElementById('draftTip');
  tip.style.display=has?'flex':'none';
  if(!has)return;
  if(draft.proj){
    const sel=document.getElementById('f-proj');
    for(const o of sel.options){ if(o.value===draft.proj||o.text===draft.proj) sel.value=o.value; }
  }
  if(draft.mt) document.getElementById('f-mt').value=draft.mt;
  if(draft.skin) document.querySelectorAll('#skinGrid span').forEach(s=>{ if(s.textContent===draft.skin) s.classList.add('on'); });
  if(draft.sens) document.querySelectorAll('#sensGrid span').forEach(s=>{ if(s.textContent===draft.sens) s.classList.add('on'); });
}
function clearDraft(){
  draft.proj=null;draft.mt=null;draft.skin=null;draft.sens=null;
  document.getElementById('f-mt').value='';
  document.querySelectorAll('#skinGrid span,#sensGrid span').forEach(s=>s.classList.remove('on'));
  document.getElementById('draftTip').style.display='none';
}
